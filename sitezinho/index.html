Pintura em pixels

canvas

Um canvas de 16 × 16. Escolha uma cor, arraste e crie um sprite. Dá até para baixar em PNG.

16 × 16 arraste para pintar
Ver o códigopintar.js ⌄
pintar.js
const tela = document.querySelector('#pixel-canvas');
const ctx = tela.getContext('2d');
let pintando = false;

tela.addEventListener('pointerdown', () => pintando = true);
tela.addEventListener('pointerup', () => pintando = false);

tela.addEventListener('pointermove', (evento) => {
  if (!pintando) return;

  const area = tela.getBoundingClientRect();
  const x = Math.floor((evento.clientX - area.left) / area.width * 16);
  const y = Math.floor((evento.clientY - area.top) / area.height * 16);

  ctx.fillStyle = '#ff5c39';
  ctx.fillRect(x, y, 1, 1);
});

Ver o códigotocar.js ⌄
tocar.js
function tocar(frequencia) {
  const agora = contexto.currentTime;
  const oscilador = contexto.createOscillator();
  const volume = contexto.createGain();

  oscilador.type = 'triangle';
  oscilador.frequency.setValueAtTime(frequencia, agora);

  volume.gain.setValueAtTime(0, agora);
  volume.gain.linearRampToValueAtTime(0.2, agora + 0.02);
  volume.gain.exponentialRampToValueAtTime(0.0001, agora + 0.9);

  oscilador.connect(volume).connect(contexto.destination);
  oscilador.start();
  oscilador.stop(agora + 1);
}

Máquina de paletas

cor

Cinco cores que combinam entre si. Clique para copiar o hexadecimal e use o cadeado para fixar suas favoritas.

paleta análoga

Clique em uma cor para copiar. O cadeado mantém a cor na próxima rodada.

Ver o códigocores.js ⌄
cores.js
function hslToHex(h, s, l) {
  s /= 100;
  l /= 100;

  const k = n => (n + h / 30) % 12;
  const a = s * Math.min(l, 1 - l);
  const f = n => l - a * Math.max(
    -1, Math.min(k(n) - 3, 9 - k(n), 1)
  );

  return '#' + [0, 8, 4]
    .map(n => Math.round(255 * f(n)).toString(16).padStart(2, '0'))
    .join('');
}

// hslToHex(12, 82, 60) → uma cor em hexadecimal
